\chapter{Maps}
\label{ch:maps}

A map assigns to each element of one set exactly one element of another. Maps are
needed \emph{before} numbers: the successor of a number, and addition and
multiplication, are all maps. Later, the \emph{functions} of calculus will be
special maps between sets of real numbers. To keep the two apart, this book says
\emph{map} for the general notion.

This chapter also fixes the one notation for maps that the whole book uses
(\cref{rule:maps}).

\section{Maps as relations}

\begin{definition}[Map]
\label{def:map}
A \emph{map} $f$ from $A$ to $B$, written $f \colon A \to B$, is a relation $f
\subseteq A \times B$ that relates every element of $A$ to exactly one element of
$B$. For a map, the relation $x \mathrel{f} y$ is written
\[
  x \via{f} y \qquad \text{(\enquote{$f$ sends $x$ to $y$})},
\]
so the defining condition reads $\forall x \in A\; \exists! y \in B\; x \via{f}
y$. That unique $y$ is called the \emph{value} of $f$ at $x$. By
\cref{rule:maps} a value is only ever named in words or by a letter introduced for
it, never by a term such as \enquote{$f$ of $x$}. $A$ is the \emph{domain} of $f$
and $B$ its \emph{codomain}.
\register{maps}{$f \colon A \to B$}{$f$ is a map from $A$ to $B$}
\register{maps}{$x \via{f} y$}{the map $f$ sends $x$ to $y$; the only notation
for values of maps}
\end{definition}

So a map \emph{is} its graph: the set of all pairs $(x, y)$ with $x \via{f} y$. A
map with domain $\N$ (\cref{ch:naturals}) is called a \emph{sequence}, and a map
with domain $I$ a \emph{family indexed by $I$}. These are only names: sequences and
families are maps and use the same notation.

\begin{theorem}[Equality of maps]
\label{thm:map-equality}
For maps $f, g \colon A \to B$:
\[
  f = g \;\Leftrightarrow\; \forall x \in A\; \forall y \in B\; (x \via{f} y
  \Rightarrow x \via{g} y).
\]
\end{theorem}

\begin{proof}
($\Rightarrow$) Substitution. ($\Leftarrow$) Let $p \in f$. Since $f \subseteq A
\times B$, $p = (x, y)$ with $x \via{f} y$. Then $x \via{g} y$, so $p \in g$. So $f
\subseteq g$. Conversely, let $p = (x, y) \in g$, and let $y'$ be the value of $f$
at $x$. Then $x \via{g} y'$ by hypothesis, so $y' = y$ because $g$ has only one
value at $x$. So $p \in f$. Use \cref{thm:subset}.
\end{proof}

\begin{theorem}[Defining maps by a rule]
\label{thm:map-rule}
Let $P(x, y)$ be a predicate such that for every $x \in A$ there is exactly one $y
\in B$ with $P(x, y)$. Then there is exactly one map $f \colon A \to B$ with
\[
  \forall x \in A\; \forall y \in B\; \bigl( x \via{f} y \Leftrightarrow P(x, y)
  \bigr).
\]
\end{theorem}

\begin{proof}
$f \coloneqq \setof{(x, y) \in A \times B}{P(x, y)}$ exists by
\cref{def:product} and is a map by the hypothesis. Uniqueness follows from
\cref{thm:map-equality}.
\end{proof}

The most common case is $P(x, y) \defiff y = t$ for an expression $t$ built from
$x$ that lies in $B$ for every $x \in A$. We then speak of \emph{the map $f \colon
A \to B$ with $x \via{f} t$ for all $x \in A$}.

\begin{definition}[Set of maps]
\label{def:map-set}
$B^A \coloneqq \setof{f \in \Pow (A \times B)}{f \text{ is a map from } A \text{
to } B}$.
\register{maps}{$B^A$}{the set of maps from $A$ to $B$}
\end{definition}

For example, $\id_A$ (\cref{def:identity}) is the map $A \to A$ with $x \via{\id_A}
x$. For any $A$ there is exactly one map $\varnothing \to A$, namely the empty
relation: the condition \enquote{for every $x \in \varnothing$} is vacuously true.

\section{Images and preimages}

\begin{definition}[Image, preimage, restriction]
\label{def:image}
Let $R$ be a relation from $A$ to $B$ and $X \subseteq A$. The \emph{image} of $X$
under $R$ is
\[
  R[X] \coloneqq \setof{y \in B}{\exists x \in X\; x \mathrel{R} y}.
\]
For a map $f \colon A \to B$ and $Y \subseteq B$ this gives the image $f[X] =
\setof{y \in B}{\exists x \in X\; x \via{f} y}$ and the \emph{preimage}
\[
  f^{-1}[Y] = \setof{x \in A}{\exists y \in Y\; x \via{f} y},
\]
the image of $Y$ under the inverse relation $f^{-1}$. The \emph{restriction} of
$f$ to $X$ is $f|_X \coloneqq f \cap (X \times B)$, a map $X \to B$ with $x
\via{f|_X} y \Leftrightarrow x \via{f} y$ for $x \in X$.
\register{maps}{$R[X]$}{image of the set $X$ under the relation $R$; for a map
$f$, $f[X]$ is the image and $f^{-1}[Y]$ the preimage}
\register{maps}{$f|_X$}{restriction of $f$ to $X$}
\end{definition}

The image of a \emph{set} is written with square brackets (\cref{rule:brackets}).
The value of a map at an \emph{element} has no bracket notation at all, so the two
can never be confused.

\section{Injective, surjective, bijective}

\begin{definition}
\label{def:injective}
A map $f \colon A \to B$ is
\begin{itemize}
  \item \emph{injective} if different elements have different values: $\forall x,
    x' \in A\; \forall y \in B\; (x \via{f} y \land x' \via{f} y \Rightarrow x =
    x')$;
  \item \emph{surjective} if every element of $B$ is a value: $\forall y \in B\;
    \exists x \in A\; x \via{f} y$, that is, $f[A] = B$;
  \item \emph{bijective} if it is both. Then every $y \in B$ is the value of
    exactly one $x \in A$.
\end{itemize}
\end{definition}

\section{Composition and inverses}

\begin{definition}[Composition]
\label{def:composition}
For a relation $R$ from $A$ to $B$ and a relation $S$ from $B$ to $C$, the
\emph{composition} is the relation from $A$ to $C$
\[
  S \circ R \coloneqq \setof{(x, z) \in A \times C}{\exists y \in B\; (x \mathrel{R}
  y \land y \mathrel{S} z)}.
\]
\register{maps}{$S \circ R$}{composition: first $R$, then $S$}
\end{definition}

\begin{theorem}[Composition of maps]
\label{thm:composition}
For maps $f \colon A \to B$ and $g \colon B \to C$, the composition $g \circ f$ is
a map $A \to C$, and
\[
  x \via{f} y \;\land\; y \via{g} z \;\Rightarrow\; x \via{g \circ f} z.
\]
\end{theorem}

\begin{proof}
Let $x \in A$, let $y$ be the value of $f$ at $x$, and $z$ the value of $g$ at
$y$. Then $x \mathrel{(g \circ f)} z$ by definition. If also $x \mathrel{(g \circ
f)} z'$, there is $y'$ with $x \via{f} y'$ and $y' \via{g} z'$. Then $y' = y$,
since $f$ has one value at $x$, and then $z' = z$, since $g$ has one value at $y$.
\end{proof}

\begin{definition}[Chains]
\label{def:chain}
A chain $x \via{f} y \via{g} z$ abbreviates $x \via{f} y \land y \via{g} z$, and
likewise for longer chains.
\register{maps}{$x \via{f} y \via{g} z$}{$x \via{f} y$ and $y \via{g} z$}
\end{definition}

\begin{theorem}
\label{thm:composition-laws}
For $f \colon A \to B$, $g \colon B \to C$ and $h \colon C \to D$:
\begin{enumerate}
  \item $h \circ (g \circ f) = (h \circ g) \circ f$ (associativity);
  \item $f \circ \id_A = f = \id_B \circ f$;
  \item if $f$ and $g$ are injective (surjective, bijective), then so is $g \circ
    f$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Both sides send $x$ to $w$ exactly when there are $y, z$ with $x \via{f} y$, $y
\via{g} z$ and $z \via{h} w$. Use \cref{thm:map-equality}.

(2) $x \via{\id_A} x \via{f} y$ and $x \via{f} y \via{\id_B} y$, so all three maps
send $x$ to the value of $f$ at $x$.

(3) Injective: let $x \via{f} y \via{g} z$ and $x' \via{f} y' \via{g} z$. Then $y =
y'$ because $g$ is injective, and then $x = x'$ because $f$ is injective.
Surjective: for $z \in C$ pick $y \in B$ with $y \via{g} z$, then $x \in A$ with $x
\via{f} y$. Then $x \via{g \circ f} z$. Bijective: combine both.
\end{proof}

\begin{theorem}[Inverse map]
\label{thm:inverse}
A map $f \colon A \to B$ is bijective if and only if there is a map $g \colon B
\to A$ with
\[
  g \circ f = \id_A \quad\text{and}\quad f \circ g = \id_B.
\]
In that case $g$ is unique and equals the inverse relation $f^{-1}$
(\cref{def:relation}), which is therefore a map, the \emph{inverse} of $f$:
\[
  y \via{f^{-1}} x \;\Leftrightarrow\; x \via{f} y.
\]
\end{theorem}

\begin{proof}
($\Rightarrow$) Since $f$ is bijective, every $y \in B$ is the value of exactly one
$x \in A$, so the inverse relation $f^{-1}$ is a map $B \to A$. If $x \via{f} y
\via{f^{-1}} x'$, then $x' \via{f} y$, so $x' = x$ by injectivity: $f^{-1} \circ f$
sends every $x$ to $x$. If $y \via{f^{-1}} x \via{f} y'$, then $x \via{f} y$, so $y'
= y$: $f \circ f^{-1}$ sends every $y$ to $y$. By \cref{thm:map-equality}, $g
\coloneqq f^{-1}$ works.

($\Leftarrow$) Injective: let $x \via{f} y$ and $x' \via{f} y$, and let $z$ be the
value of $g$ at $y$. Then $x \via{g \circ f} z$ and $x' \via{g \circ f} z$
(\cref{thm:composition}). Since $g \circ f = \id_A$, $z = x$ and $z = x'$.
Surjective: let $y \in B$. Since $f \circ g = \id_B$, $y \via{f \circ g} y$, so
there is $x$ with $y \via{g} x \via{f} y$.

Uniqueness: if $g'$ also works, then by \cref{thm:composition-laws}
\[
  g' = g' \circ \id_B = g' \circ (f \circ g) = (g' \circ f) \circ g = \id_A \circ
  g = g.
\]
Since $f^{-1}$ works, every such $g$ equals $f^{-1}$.
\end{proof}

\section{Maps on quotient sets}

When we construct $\Z$ and $\Q$, arithmetic is first defined on representatives
and then moved to classes. The following theorem makes that precise.

\begin{theorem}[Inducing a map on a quotient]
\label{thm:quotient-map}
Let $\sim$ be an equivalence relation on $A$ and $f \colon A \to B$ a map that
respects $\sim$:
\[
  a \sim a' \land a \via{f} b \land a' \via{f} b' \;\Rightarrow\; b = b'.
\]
Then there is exactly one map $\bar f \colon A/{\sim} \to B$ with
\[
  a \via{f} b \;\Rightarrow\; [a] \via{\bar f} b \qquad \text{for all } a \in A,\;
  b \in B.
\]
\end{theorem}

\begin{proof}
For a class $C \in A/{\sim}$ we need exactly one $b$ with $a \via{f} b$ for some $a
\in C$. There is at least one, because $C$ is nonempty (\cref{thm:partition}).
There is at most one: if $a, a' \in C$, then $a \sim a'$ (\cref{lem:classes}), so
their values agree. Now apply \cref{thm:map-rule}. Uniqueness follows from
\cref{thm:map-equality}, because every class is of the form $[a]$.
\end{proof}

The hypothesis \enquote{$f$ respects $\sim$} is exactly the statement that $\bar
f$ is well-defined (\cref{def:well-defined}).

\section{Operations}

\begin{definition}[Operation]
\label{def:operation}
A \emph{binary operation} on a set $A$ is a map $\ast \colon A \times A \to A$
that has been given a symbol of its own. Its value is written as a term
(\cref{rule:operations}): $a \ast b$ denotes the value of $\ast$ at $(a, b)$, so
\[
  (a, b) \via{\ast} c \;\Leftrightarrow\; a \ast b = c.
\]
Likewise a \emph{unary operation} is a map $A \to A$ with a symbol of its own,
whose value at $a$ is written with that symbol, as in $-a$. An operation $\ast$ is
\emph{associative} if $(a \ast b) \ast c = a \ast (b \ast c)$ for all $a, b, c$,
and \emph{commutative} if $a \ast b = b \ast a$ for all $a, b$. An element $e \in
A$ is \emph{neutral} for $\ast$ if $e \ast a = a = a \ast e$ for all $a$.
\register{maps}{$a \ast b$}{value of the binary operation $\ast$ at $(a, b)$;
operations are the only maps whose values are terms}
\end{definition}

Constructions that apply to all sets, such as $\cup$ and $\circ$, are written in the
same way. A neutral element is unique when it exists: if $e$ and $e'$ are both
neutral, then $e = e \ast e' = e'$.

\begin{exercises}
  \item Let $f \colon A \to B$ and $X, X' \subseteq A$. Show $f[X \cup X'] = f[X]
    \cup f[X']$ and $f[X \cap X'] \subseteq f[X] \cap f[X']$. Find an example where
    the inclusion is strict. Show it is an equality when $f$ is injective.
  \item Show that preimages preserve everything: $f^{-1}[Y \cap Y'] = f^{-1}[Y]
    \cap f^{-1}[Y']$, and the same for $\cup$ and $\setminus$.
  \item If $g \circ f$ is injective, must $f$ be injective? Must $g$ be?
  \item Show: for $A \neq \varnothing$, $f \colon A \to B$ is injective if and only
    if there is a $g \colon B \to A$ with $g \circ f = \id_A$.
\end{exercises}
