% ---------------------------------------------------------------------------
% Register of notation.
%
%   \register{model}{notation}{meaning}
%
% placed inside the definition that introduces a notation, records it for the
% register at the end of the book, with a link back to that definition.
% Entries are written to the .aux file and read back on the next run, so the
% register is always generated from the text itself. Models are listed in the
% order declared here.
% ---------------------------------------------------------------------------
\makeatletter
\newcommand{\declaremodel}[2]{%
  \listgadd{\reg@models}{#1}%
  \csgdef{reg@title@#1}{#2}%
  \csgdef{reg@rows@#1}{}}
\newcommand{\register}[3]{%
  \immediate\write\@auxout{\string\regentry{#1}%
    {\unexpanded{#2}}{\unexpanded{#3}}{\thetheorem}{\@currentHref}}}
\newcommand{\regentry}[5]{%
  \ifcsdef{reg@title@#1}{}{\PackageError{autodiductive}%
    {Unknown model `#1' in \string\register}{Declare it with \string\declaremodel.}}%
  \csgappto{reg@rows@#1}{#2 & #3 & \hyperlink{#5}{#4} \\}}
\newcommand{\reg@print}[1]{%
  \section*{\csuse{reg@title@#1}}
  \begin{longtable}{@{}p{0.28\textwidth}p{0.55\textwidth}r@{}}
    \toprule
    Notation & Meaning & Defined in \\
    \midrule
    \endhead
    \csuse{reg@rows@#1}
    \bottomrule
  \end{longtable}}
\newcommand{\printregister}{\forlistloop{\reg@print}{\reg@models}}
\makeatother
